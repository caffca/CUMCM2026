% ---------- 字体（fontspec + ctex，xelatex 编译；本机字体自动回退，禁绝对路径） ----------
% 与 Typst 模板一致的字体角色（引擎无关语义）：正文宋体 / 标题黑体 / 强调楷体 / 西文 Times /
% 代码等宽 Consolas→Courier New→Latin Modern Mono→Menlo（Windows 优先，避免 Men.cfg 配置告警）；
% 中文代码字体 FangSong→SimSun（TPL-L08 修复 \CJKttdefault 未定义）。
% TPL-L07：中文粗体/斜体映射（正文宋体粗体映射黑体、斜体映射楷体），
% 禁止本机绝对字体路径；\IfFontExistsTF 探测失败时回退 macOS 族。
\IfFontExistsTF{SimSun}{%
  \setCJKmainfont{SimSun}[BoldFont=SimHei, ItalicFont=KaiTi]}{%
  \setCJKmainfont{Songti SC}[BoldFont=Heiti SC, ItalicFont=Kaiti SC]}
\IfFontExistsTF{SimSun}{%
  \newCJKfontfamily\songti{SimSun}[BoldFont=SimHei, ItalicFont=KaiTi]}{%
  \newCJKfontfamily\songti{Songti SC}[BoldFont=Heiti SC, ItalicFont=Kaiti SC]}
\IfFontExistsTF{SimHei}{%
  \newCJKfontfamily\heiti{SimHei}[BoldFont=SimHei, ItalicFont=KaiTi]}{%
  \newCJKfontfamily\heiti{Heiti SC}[BoldFont=Heiti SC, ItalicFont=Kaiti SC]}
\IfFontExistsTF{KaiTi}{%
  \newCJKfontfamily\kaishu{KaiTi}[BoldFont=SimHei, ItalicFont=KaiTi]}{%
  \newCJKfontfamily\kaishu{Kaiti SC}[BoldFont=Heiti SC, ItalicFont=Kaiti SC]}
\IfFontExistsTF{Times New Roman}{\setmainfont{Times New Roman}}{}
% TPL-L06：Windows 优先等宽回退链（Consolas→Courier New→Latin Modern Mono），Menlo 仅兜底
\IfFontExistsTF{Consolas}{\setmonofont{Consolas}}{%
  \IfFontExistsTF{Courier New}{\setmonofont{Courier New}}{%
    \IfFontExistsTF{Latin Modern Mono}{\setmonofont{Latin Modern Mono}}{%
      \IfFontExistsTF{Menlo}{\setmonofont{Menlo}}{}}}}
% TPL-L08：CJK 代码字体（\CJKttdefault 未定义告警消除）；Windows 优先 FangSong→SimSun
\IfFontExistsTF{FangSong}{\setCJKmonofont{FangSong}}{%
  \IfFontExistsTF{SimSun}{\setCJKmonofont{SimSun}}{}}
